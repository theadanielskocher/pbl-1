        \begin{algorithm}[H]
        	\footnotesize 
            \caption{GetPathToRoot(Node)} \label{alg:getpathtoroot}
            \begin{algorithmic}
                \State Khởi tạo mảng Path rỗng
                \While{Node không phải là rỗng}
                    \State Thêm Node vào đầu mảng Path (để Gốc luôn ở vị trí số 0)
                    \State Node = Node.parent
                \EndWhile
                \State \Return Path
            \end{algorithmic}
        \end{algorithm}
